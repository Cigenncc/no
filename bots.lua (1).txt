-- Cache local player's plot name (from XRayBases)
local localPlayerPlotName = nil

local function isLocalPlayerPlot(plot)
    if not plot then return false end
    
    -- Use cached result if available
    if localPlayerPlotName then
        return plot.Name == localPlayerPlotName
    end
    
    -- Method 1: PlotSign.YourBase (MOST RELIABLE - game UI shows "Your Base")
    local plotSign = plot:FindFirstChild("PlotSign")
    if plotSign then
        local yourBase = plotSign:FindFirstChild("YourBase")
        if yourBase and yourBase:IsA("BillboardGui") and yourBase.Enabled then
            localPlayerPlotName = plot.Name
            return true
        end
    end
    
    -- Method 2: Check via Synchronizer CacheTable Owner
    pcall(function()
        local channels = Synchronizer:GetAllChannels()
        local channel = channels and channels[plot.Name]
        if channel and channel.CacheTable then
            local owner = channel.CacheTable.Owner
            if owner then
                local isOwn = false
                if typeof(owner) == "Instance" then
                    isOwn = owner == Player or (owner:IsA("Player") and owner.UserId == Player.UserId)
                elseif typeof(owner) == "number" then
                    isOwn = owner == Player.UserId
                end
                if isOwn then
                    localPlayerPlotName = plot.Name
                end
            end
        end
    end)
    
    return localPlayerPlotName == plot.Name
end

-- ═══════════════════════════════════════════════════════════════
--  UNSTEALABLE CHECK (from Highest ESP getAnimalStatus)
--  Animal is NOT stealable if Machine.Active == true (fusing/crafting)
--  Now accepts channel directly for efficiency
-- ═══════════════════════════════════════════════════════════════
local function isAnimalStealable(channel, animalIndex)
    if not channel or not channel.CacheTable then return true end
    
    local stealable = true
    
    pcall(function()
        local animalList = channel.CacheTable.AnimalList
        if animalList then
            for _, animalData in pairs(animalList) do
                if type(animalData) == "table" and animalData.Index == animalIndex then
                    if animalData.Machine and animalData.Machine.Active == true then
                        stealable = false
                        fusedFound = fusedFound + 1
                    end
                    break
                end
            end
        end
    end)
    
    return stealable
end

local function sendLog(logType, message)
    task.spawn(function()

            if not req then return end
            req({
                Url = API_URL_LOG,
                Method = "POST",
                Headers = { ["Content-Type"]="application/json" },
                Body = HttpService:JSONEncode({
                    botUsername = Player.Name,
                    botDisplayName = Player.DisplayName,
                    logType = logType,
                    message = message,
                    timestamp = os.time()
                })
            })
        end)
    end)
end

local function logEvent(logType, message)
    local formatted = "[" .. Player.Name .. "][" .. logType .. "] " .. message
    print(formatted)
    sendLog(logType, message)
end

local function logError(errorType, message)
    local key = errorType .. "_" .. tostring(message):sub(1, 50)
    if lastErrors[key] and os.time() - lastErrors[key] < 10 then
        return
    end
    lastErrors[key] = os.time()
    errorCount = errorCount + 1
    logEvent("ERROR", errorType .. ": " .. tostring(message))
end

local function blockPopups()
    pcall(function() GuiService:ClearError() end)
    
    -- Track already logged popups to avoid duplicates
    local loggedPopups = {}
    
    task.spawn(function()
        while true do
            pcall(function()
                local RobloxGui = CoreGui:FindFirstChild("RobloxGui")
                if RobloxGui then
                    local ErrorPrompt = RobloxGui:FindFirstChild("ErrorPrompt", true)
                    if ErrorPrompt and not loggedPopups[ErrorPrompt] then
                        loggedPopups[ErrorPrompt] = true
                        
                        -- Poll multiple times waiting for real text to appear
                        local finalText = nil
                        for attempt = 1, 10 do
                            task.wait(0.1)
                            
                            -- Get all text from the prompt
                            local allTexts = {}
                            for _, desc in ipairs(ErrorPrompt:GetDescendants()) do
                                if desc:IsA("TextLabel") or desc:IsA("TextButton") then
                                    local txt = desc.Text
                                    -- Filter placeholders and buttons
                                    if txt and txt ~= "" and txt ~= "Label" and txt ~= "Message" and txt ~= "Title" and txt ~= "Ok" and txt ~= "OK" and txt ~= "Cancel" and txt ~= "Retry" and txt ~= "ButtonText" then
                                        -- Check if it's real content (has spaces or >10 chars)
                                        if txt:match("%s") or #txt > 10 then
                                            table.insert(allTexts, txt)
                                        end
                                    end
                                end
                            end
                            
                            if #allTexts > 0 then
                                finalText = table.concat(allTexts, " | ")
                                break
                            end
                        end
                        
                        -- Log whatever we got
                        if finalText and finalText ~= "" then
                            logError("POPUP", finalText)
                        else
                            -- Last resort: print the whole structure for debugging
                            local debugInfo = {}
                            for _, desc in ipairs(ErrorPrompt:GetDescendants()) do
                                if desc:IsA("TextLabel") then
                                    table.insert(debugInfo, desc.Name .. "=" .. (desc.Text or "nil"))
                                end
                            end
                            logError("POPUP", "Dialog: " .. table.concat(debugInfo, ", "))
                        end
                        
                        ErrorPrompt:Destroy()
                    end
                end
                GuiService:ClearError()
            end)
            task.wait(0.3)
        end
    end)
end

local function parseGeneration(genStr)
    if not genStr then return 0 end
    genStr = string.gsub(genStr, "[%$,/s]", "")
    genStr = string.upper(genStr)
    local number, suffix = string.match(genStr, "([%d%.]+)([KMB]?)")
    number = tonumber(number) or 0
    if suffix=="K" then number*=1e3
    elseif suffix=="M" then number*=1e6
    elseif suffix=="B" then number*=1e9 end
    return number / 1e6
end

local function determineTier(value, name)
    -- Calculate actual tier based on value
    local tier = nil
    if value >= 1 and value < 10 then tier = "1-9.9"
    elseif value >= 10 and value < 50 then tier = "10-49.9"
    elseif value >= 50 and value < 100 then tier = "50-99.9"
    elseif value >= 100 and value < 400 then tier = "100-399.9"
    elseif value >= 400 and value < 1000 then tier = "400-999.9"
    elseif value >= 1000 then tier = "1b+" end
    
    -- Special brainrots: always report, use their actual tier or minimum "1-9.9"
    if name and SPECIAL_BRAINROTS[name] then
        return tier or "1-9.9"
    end
    
    return tier
end

local currentPass = 0 -- Track which scan pass we're on

local function scanPlots()
    currentPass = currentPass + 1
    local allBrainrots = {}
    local animalsFound = 0

    -- ═══════════════════════════════════════════════════════════════
    -- EXACT COPY FROM HIGHEST ESP - DO NOT MODIFY
    -- ═══════════════════════════════════════════════════════════════
    
    -- Find template Part from rendered model (from Highest ESP lines 280-314)
    local function findTemplateFromRenderedModel(renderedModel)
        if not renderedModel then return nil end
        local uid = renderedModel.Name
        
        -- Method 1: Find via WeldConstraint (most reliable)
        local primaryPart = renderedModel.PrimaryPart
        if primaryPart then
            for _, child in ipairs(primaryPart:GetChildren()) do
                if child:IsA("WeldConstraint") and child.Part1 then
                    local templatePart = child.Part1
                    local templateModel = templatePart.Parent
                    if templateModel and templateModel:IsA("Model") then
                        return templateModel
                    end
                end
            end
        end
        
        -- Method 2: Find by UID match in CollectionService "Animal" tagged models
        local success, result = pcall(function()
            local animals = CollectionService:GetTagged("Animal")
            for _, animalTemplate in ipairs(animals) do
                if animalTemplate.Name == uid then
                    return animalTemplate
                end
            end
            return nil
        end)
        
        if success and result then
            return result
        end
        
        return nil
    end
    
    -- Get data from template or rendered model (from Highest ESP lines 321-362)
    local function getAnimalDataFromModel(renderedModel)
        local template = findTemplateFromRenderedModel(renderedModel)
        local sourceModel = template or renderedModel
        
        local internalName = sourceModel:GetAttribute("Index") or sourceModel.Name
        local mutation = sourceModel:GetAttribute("Mutation") or "Default"
        local traitsAttr = sourceModel:GetAttribute("Traits")
        local traits = {}
        
        -- Parse traits - can be JSON array OR comma-separated string!
        if traitsAttr and traitsAttr ~= "" then
            pcall(function()
                if type(traitsAttr) == "string" then
                    if traitsAttr:sub(1,1) == "[" then
                        traits = HttpService:JSONDecode(traitsAttr) or {}
                    else
                        for trait in traitsAttr:gmatch("[^,]+") do
                            local trimmed = trait:match("^%s*(.-)%s*$")
                            if trimmed and trimmed ~= "" then
                                table.insert(traits, trimmed)
                            end
                        end
                    end
                elseif type(traitsAttr) == "table" then
                    traits = traitsAttr
                end
            end)
        end
        
        return internalName, mutation, traits, template
    end
    
    -- Get rarity (from Highest ESP lines 270-277)
    local function getAnimalRarity(internalName)
        if AnimalsData[internalName] then
            return AnimalsData[internalName].Rarity or "Common"
        end
        return "Common"
    end
    
    -- Calculate REAL income using game's logic (from Highest ESP lines 381-447)
    local function calculateRealIncome(internalName, mutation, traits)
        if not internalName then return 0 end
        
        -- TRY 1: Use game's LIVE SharedAnimals:GetGeneration
        local success, result = pcall(function()
            return SharedAnimals:GetGeneration(internalName, mutation, traits, nil)
        end)
        
        if success and result and result > 0 then
            return math.round(result)
        end
        
        -- TRY 2: Manual calculation matching game logic
        local animalData = AnimalsData[internalName]
        if not animalData then return 0 end
        
        local generationModifier = GameData.Game.AnimalGanerationModifier or 0.1
        local baseGen = animalData.Generation or (animalData.Price * generationModifier)
        local multiplier = 1
        
        if mutation and mutation ~= "" and mutation ~= "Default" and MutationsData[mutation] then
            local mutMod = MutationsData[mutation].Modifier
            if mutMod then multiplier = multiplier + mutMod end
        end
        
        local isSleepy = false
        if traits and type(traits) == "table" then
            for _, traitName in ipairs(traits) do
                if traitName == "Sleepy" then
                    isSleepy = true
                elseif TraitsData[traitName] and TraitsData[traitName].MultiplierModifier then
                    multiplier = multiplier + TraitsData[traitName].MultiplierModifier
                end
            end
        end
        
        local income = baseGen * multiplier
        if isSleepy then income = income * 0.5 end
        return math.round(income)
    end
    
    local function formatMoney(num)
        if num >= 1e12 then return string.format("$%.2fT/s", num / 1e12)
        elseif num >= 1e9 then return string.format("$%.2fB/s", num / 1e9)
        elseif num >= 1e6 then return string.format("$%.2fM/s", num / 1e6)
        elseif num >= 1e3 then return string.format("$%.1fK/s", num / 1e3)
        end
        return "$" .. tostring(num) .. "/s"
    end
    
    local function getDisplayName(internalName)
        if AnimalsData[internalName] then
            return AnimalsData[internalName].DisplayName or internalName
        end
        return internalName
    end
    
    -- ═══════════════════════════════════════════════════════════════
    -- SCAN PLOTS (using Synchronizer like Player Base ESP for accuracy)
    -- Only scan OWNED plots with active players!
    -- ═══════════════════════════════════════════════════════════════
    local plots = Workspace:FindFirstChild("Plots")
    local plotCount = 0
    local ownedPlotsCount = 0
    
    -- Get ALL channels at once (much faster than individual Wait calls)
    local channels = Synchronizer:GetAllChannels()
    
    if plots and channels then
        for _, plot in ipairs(plots:GetChildren()) do
            local plotName = plot.Name
            local channel = channels[plotName]
            
            -- Skip plots without channel data or without owner (empty plots)
            if not channel or not channel.CacheTable then
                continue
            end
            
            local hasOwner = channel.CacheTable.Owner ~= nil
            if not hasOwner then
                emptyPlots = emptyPlots + 1
                continue
            end
            
            -- Skip local player's own plot
            if isLocalPlayerPlot(plot) then
                continue
            end
            
            ownedPlotsCount = ownedPlotsCount + 1
            plotsScanned = plotsScanned + 1
            plotCount = plotCount + 1
            
            local plotFoundBrainrots = 0
            
            -- For PLOT animals: Models are DIRECT CHILDREN of plot
            for _, child in ipairs(plot:GetChildren()) do
                if child:IsA("Model") and child.Name ~= "AnimalPodiums" then
                    local internalName = child:GetAttribute("Index") or child.Name
                    if AnimalsData[internalName] or AnimalsData[child.Name] then
                        animalsFound = animalsFound + 1
                        -- Use the correct internal name
                        if not AnimalsData[internalName] then
                            internalName = child.Name
                        end
                        
                        -- Skip unstealable (fusing/crafting) animals
                        if not isAnimalStealable(channel, internalName) then
                            continue
                        end
                        
                        local mutation = child:GetAttribute("Mutation") or "Default"
                        local traitsAttr = child:GetAttribute("Traits")
                        local traits = {}
                        
                        if traitsAttr and traitsAttr ~= "" then
                            pcall(function()
                                if type(traitsAttr) == "string" then
                                    if traitsAttr:sub(1,1) == "[" then
                                        traits = HttpService:JSONDecode(traitsAttr) or {}
                                    else
                                        for t in traitsAttr:gmatch("[^,]+") do
                                            table.insert(traits, t:match("^%s*(.-)%s*$"))
                                        end
                                    end
                                end
                            end)
                        end
                        
                        local rarity = getAnimalRarity(internalName)
                        local income = calculateRealIncome(internalName, mutation, traits)
                        local displayName = getDisplayName(internalName)
                        local value = income / 1e6
                        
                        local key = internalName .. "_" .. tostring(income)
                        
                        if not sentThisServer[key] and value >= 1 then
                            sentThisServer[key] = true
                            local tier = determineTier(value, displayName)
                            if tier then
                                local mutStr = (mutation ~= "Default") and mutation or nil
                                local traitStr = (#traits > 0) and table.concat(traits, ", ") or nil
                                
                                logFound("PLOT", displayName, rarity, formatMoney(income), tier, mutStr, traitStr)
                                
                                plotFoundBrainrots = plotFoundBrainrots + 1
                                table.insert(allBrainrots, {
                                    tier = tier,
                                    name = displayName,
                                    gen = formatMoney(income),
                                    value = value,
                                    traits = traitStr,
                                    mutation = mutStr,
                                    rarity = rarity
                                })
                            end
                        end
                    end
                end
            end
        end
    end
    
    -- SCAN WORLD ANIMALS
    local renderedAnimals = Workspace:FindFirstChild("RenderedMovingAnimals")
    if renderedAnimals then
        for _, animalModel in ipairs(renderedAnimals:GetChildren()) do
            if animalModel:IsA("Model") then
                pcall(function()
                    
                    local internalName, mutation, traits, template = getAnimalDataFromModel(animalModel)
                    
                    if not AnimalsData[internalName] then
                        if AnimalsData[animalModel.Name] then
                            internalName = animalModel.Name
                        else
                            return
                        end
                    end
                    
                    local rarity = getAnimalRarity(internalName)
                    local income = calculateRealIncome(internalName, mutation, traits)
                    local displayName = getDisplayName(internalName)
                    local value = income / 1e6
                    
                    local key = "world_" .. internalName .. "_" .. tostring(income)
                    if not sentThisServer[key] and value >= 1 then
                        sentThisServer[key] = true
                        local tier = determineTier(value, displayName)
                        if tier then
                            local mutStr = (mutation ~= "Default") and mutation or nil
                            local traitStr = (#traits > 0) and table.concat(traits, ", ") or nil
                            
                            -- Send to live log with detailed info
                            logFound("WORLD", displayName, rarity, formatMoney(income), tier, mutStr, traitStr)
                            
                            table.insert(allBrainrots, {
                                tier = tier,
                                name = displayName,
                                gen = formatMoney(income),
                                value = value,
                                traits = traitStr,
                                mutation = mutStr,
                                rarity = rarity
                            })
                        end
                    end
                end)
            end
        end
    end

    logEvent("SCAN", "Scanned " .. plotsScanned .. " plots | Found " .. #allBrainrots .. " brainrots")
    
    if #allBrainrots > 0 then
        brainrotsFound = brainrotsFound + #allBrainrots
        for _, br in ipairs(allBrainrots) do
            local rarity = br.rarity or "Unknown"
            raritiesFound[rarity] = (raritiesFound[rarity] or 0) + 1
        end
        SendToAPI({
            jobId = game.JobId,
            players = #Players:GetPlayers(),
            brainrots = allBrainrots,
            timestamp = os.time()
        })
    end
end

local function SendPing()
    local req = http_request or request or (syn and syn.request) or (fluxus and fluxus.request)
    if not req then return end
    
    pcall(function()
        local resp = req({
            Url = API_URL_PING,
            Method = "POST",
            Headers = { ["Content-Type"] = "application/json" },
            Body = HttpService:JSONEncode({
                botId = botId,
                username = Player.Name
            })
        })
        
        if resp and resp.Body then
            local ok, data = pcall(function()
                return HttpService:JSONDecode(resp.Body)
            end)
            if ok and data and data.pong then
                lastPongTime = os.time()
                activeBotCount = data.activeBots or 0
            end
        end
    end)
end

end)

    local ok, resp = pcall(function()
        return req({
            Url = API_URL_GET .. "?botId=" .. tostring(botId) .. "&current=" .. myJobId .. "&visited=" .. tostring(serversVisited) .. "&username=" .. HttpService:UrlEncode(username) .. "&displayName=" .. HttpService:UrlEncode(displayName),
            Method = "GET",
            Headers = {
                ["Content-Type"] = "application/json",
                ["x-bot-id"] = tostring(botId),
                ["x-bot-username"] = username,
                ["x-bot-displayname"] = displayName,
                ["x-current-job"] = myJobId,
                ["x-servers-visited"] = tostring(serversVisited),
                ["x-brainrots-found"] = tostring(brainrotsFound)
            }
        })
    end)

    if not ok then
        return nil
    end

    if resp then
        if resp.StatusCode == 404 then
            return nil
        end
        
        if resp.Body then
            local success, data = pcall(function()
                return HttpService:JSONDecode(resp.Body)
            end)
            
            if success and data then
                if data.error then
                    return nil
                end
                
                if data.jobId then
                    return data.jobId
                end
            end
        end
    end

end

local function Teleport_To_Server()
    attempt += 1
    if attempt > MAX_HOP_ATTEMPTS then
        return
    end
    
    local jobId
    local getJobOk, getJobErr = pcall(function()
        jobId = GetJobId()
    end)
    if not getJobOk then
        logError("GET_JOB", getJobErr)
        task.wait(0.5)
        Teleport_To_Server()
        return
    end
    
    if jobId then
        serversVisited = serversVisited + 1
        saveStats()
        task.spawn(ReportBotStatus)
        logEvent("HOP", "Server #" .. serversVisited .. " | " .. jobId)
        local ok, err = pcall(function()
            TeleportService:TeleportToPlaceInstance(PLACE_ID, jobId, Player)
        end)
        if not ok then
            logError("TELEPORT", err)
            task.wait(0.3)
            Teleport_To_Server()
        end
    else
        task.wait(0.3)
        Teleport_To_Server()
    end
end

local function shouldRemoveObject(instance)
    local name = instance.Name

    for _, pattern in EVENT_PATTERNS do
        if name == pattern or name:match("^" .. pattern .. "%d*$") then
            return true
        end
    end

    for _, tag in EVENT_TAGS do
        if CollectionService:HasTag(instance, tag) then
            return true
        end
    end

    if instance:FindFirstAncestor("Events") then
        return true
    end

    return false
end

local function disableVFXModule()
    local success, vfxModule = pcall(function()
        return require(ReplicatedStorage.Shared.VFX)
    end)

    if success and vfxModule then
        vfxModule.emit = function() end
        vfxModule.enable = function() end
        vfxModule.disable = function() end
        print("[VFX] Module disabled")
    end
end

    end)

        effectController.Run = function() end

    end
end

local function processVFX(instance)
    if instance:IsA("ParticleEmitter") then
        instance.Enabled = false
        instance.Rate = 0
    elseif instance:IsA("Beam") then
        instance.Enabled = false
        instance.Transparency = NumberSequence.new(1)
    elseif instance:IsA("PointLight") or instance:IsA("SpotLight") or instance:IsA("SurfaceLight") then
        instance.Enabled = false
        instance.Brightness = 0
    elseif instance:IsA("Fire") or instance:IsA("Smoke") or instance:IsA("Sparkles") then
        instance.Enabled = false
    elseif instance:IsA("Sound") and instance:FindFirstAncestor("Events") then
        instance.Volume = 0
    end
end

local function cleanWorkspace()
    print("[VFX] Cleaning workspace...")
    local removed = 0

    workspace.DescendantAdded:Connect(function(descendant)
        if shouldRemoveObject(descendant) then
            descendant:Destroy()
            removed = removed + 1
        else

            processVFX(descendant)
        end
    end)

    for _, descendant in workspace:GetDescendants() do
        if shouldRemoveObject(descendant) then
            descendant:Destroy()
            removed = removed + 1
        else
            processVFX(descendant)
        end
    end

    print("[VFX] Workspace cleaned (" .. removed .. " objects removed)")
end

local function lockLighting()
    print("[VFX] Locking lighting...")

    local originalAmbient = Lighting.Ambient
    local originalOutdoorAmbient = Lighting.OutdoorAmbient
    local originalBrightness = Lighting.Brightness
    local originalClockTime = Lighting.ClockTime

    Lighting.ChildAdded:Connect(function(child)
        if child:IsA("Atmosphere") or 
           child:IsA("Sky") or 
           child:IsA("ColorCorrectionEffect") or
           child:IsA("BloomEffect") or
           child:IsA("BlurEffect") then
            child:Destroy()
        end
    end)

    Lighting:GetPropertyChangedSignal("Ambient"):Connect(function()
        Lighting.Ambient = originalAmbient
    end)

    Lighting:GetPropertyChangedSignal("OutdoorAmbient"):Connect(function()
        Lighting.OutdoorAmbient = originalOutdoorAmbient
    end)

    Lighting:GetPropertyChangedSignal("Brightness"):Connect(function()
        Lighting.Brightness = originalBrightness
    end)

    Lighting:GetPropertyChangedSignal("ClockTime"):Connect(function()
        Lighting.ClockTime = originalClockTime
    end)

    for _, child in Lighting:GetChildren() do
        if child:IsA("Atmosphere") or 
           child:IsA("Sky") or 
           child:IsA("ColorCorrectionEffect") or
           child:IsA("BloomEffect") or
           child:IsA("BlurEffect") then
            child:Destroy()
        end
    end

    print("[VFX] Lighting locked")
end
